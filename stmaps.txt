<script>
function calculateTotal() {
    let rows = document.querySelectorAll("tbody tr");
    let totalAmount = 0;

    // Loop through stamp value rows (Skipping header row index 0)
    for (let i = 1; i <= 2; i++) {
        let row = document.querySelectorAll("table")[0].rows[i];
        let val = parseFloat(row.querySelector(".stamp-value").value) || 0;
        let qty = parseFloat(row.querySelector(".qty").value) || 0;
        
        let amt = val * qty;
        row.querySelector(".amount").value = amt.toFixed(2);
        totalAmount += amt;
    }

    // Calculate 3% discount and final total
    let discount = totalAmount * 0.03;
    let netTotal = totalAmount - discount;

    document.getElementById("grossAmount").innerText = totalAmount.toFixed(2);
    document.getElementById("discount").innerText = discount.toFixed(2);
    document.getElementById("netTotal").innerText = netTotal.toFixed(2);
}
</script>
